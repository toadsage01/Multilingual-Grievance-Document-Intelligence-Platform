name: CI

on:
  push:
    branches: [main]
  pull_request:
    branches: [main]

jobs:
  lint:
    runs-on: ubuntu-latest
    steps:
      - uses: actions/checkout@v4
      - uses: actions/setup-python@v5
        with:
          python-version: "3.12"
      - name: Install deps
        run: |
          python -m pip install --upgrade pip
          pip install ruff==0.6.9
      - name: Lint
        run: ruff check apps core config tests

  test:
    runs-on: ubuntu-latest

    services:
      postgres:
        image: pgvector/pgvector:pg16
        env:
          POSTGRES_USER: setu
          POSTGRES_PASSWORD: setu
          POSTGRES_DB: setu
        ports:
          - 5432:5432
        options: >-
          --health-cmd "pg_isready -U setu -d setu"
          --health-interval 5s
          --health-timeout 5s
          --health-retries 10

      redis:
        image: redis:7-alpine
        ports:
          - 6379:6379
        options: >-
          --health-cmd "redis-cli ping"
          --health-interval 5s
          --health-timeout 5s
          --health-retries 10

    env:
      DJANGO_SETTINGS_MODULE: config.settings.dev
      DJANGO_SECRET_KEY: ci-test-secret-only-do-not-use-in-prod
      DJANGO_DEBUG: "True"
      DATABASE_URL: postgresql://setu:setu@localhost:5432/setu
      REDIS_URL: redis://localhost:6379/0
      GROQ_API_KEY: ""
      GEMINI_API_KEY: ""

    steps:
      - uses: actions/checkout@v4
      - uses: actions/setup-python@v5
        with:
          python-version: "3.12"
      - name: Install deps
        run: |
          python -m pip install --upgrade pip
          pip install -r requirements.txt
      - name: Run migrations
        run: python manage.py migrate
      - name: Run unit tests
        run: python -m pytest tests/unit -v
      - name: Run integration tests
        run: python -m pytest tests/integration -v

  build:
    runs-on: ubuntu-latest
    needs: [lint, test]
    steps:
      - uses: actions/checkout@v4
      - uses: actions/setup-python@v5
        with:
          python-version: "3.12"
      - name: Verify app loads
        run: |
          python -m pip install --upgrade pip
          pip install Django==5.0.7 djangorestframework==3.15.2 drf-spectacular==0.27.2 django-environ==0.11.2 "psycopg[binary]==3.2.1" pgvector==0.3.0 django-redis==5.4.0 django-rq==2.10.2
          DJANGO_SETTINGS_MODULE=config.settings.prod python manage.py check
